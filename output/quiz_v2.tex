\question L'intégrale \( \int_0^1 x^2\,dx \) est égale à:

%Vérification de la présence d'une image

%Type de question - vrai ou faux

\begin{choices}

  
    \choice $\tfrac{1}{4}$
  

  
    \choice $1$
  

  
    \CorrectChoice $\tfrac{1}{3}$
  

\end{choices}

%Texte de la question
\question La dérivée de \( F(x) = \int_0^x \sin t\,dt \) est:

%Vérification de la présence d'une image

%Type de question - vrai ou faux

\begin{choices}

  
    \choice $\cos x$
  

  
    \choice $-\sin x$
  

  
    \CorrectChoice $\sin x$
  

\end{choices}
